! =============================================================
! SW-CORE  -  Layer-3 Core/Distribution Switch
! Role: Inter-VLAN routing (SVIs), DHCP server for all department
!       VLANs, central enforcement point for the Student/Wireless
!       restriction ACL, spanning-tree root for every VLAN
! Platform: Cisco Catalyst 3560-24PS (multilayer switch)
! =============================================================

hostname SW-CORE
!
ip routing
!
no ip domain-lookup
ip domain-name college.local
enable secret Cl@ssC0llege!Core
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY - College Core Switch (SW-CORE)
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Core
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Core
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
snmp-server community COLLEGE-RO RO
!
! -------------------------------------------------------------
! VLAN Database
! -------------------------------------------------------------
vlan 10
 name ADMIN
vlan 20
 name STUDENTS
vlan 25
 name WIRELESS
vlan 30
 name FACULTY
vlan 40
 name IT_DEPT
vlan 50
 name LIBRARY
vlan 60
 name ACCOUNTS
vlan 70
 name MGMT_PRINCIPAL
vlan 80
 name SERVERS
vlan 99
 name NET_MGMT
vlan 999
 name UNUSED_PARKING
!
! -------------------------------------------------------------
! Spanning Tree - SW-CORE is root for every VLAN
! -------------------------------------------------------------
spanning-tree mode pvst
spanning-tree vlan 10,20,25,30,40,50,60,70,80,99 priority 4096
!

 no shutdown
!

! -------------------------------------------------------------
! Extended ACL - Students/Wireless restricted from Accounts & Mgmt
! Applied inbound on Vlan20 and Vlan25 SVIs above
! -------------------------------------------------------------
ip access-list extended STUDENT-RESTRICT
 remark Deny Students/Wireless -> sensitive Accounts & Management subnets
 deny   ip 10.10.20.0 0.0.0.255 10.10.60.0 0.0.0.255
 deny   ip 10.10.20.0 0.0.0.255 10.10.70.0 0.0.0.255
 deny   ip 10.10.25.0 0.0.0.255 10.10.60.0 0.0.0.255
 deny   ip 10.10.25.0 0.0.0.255 10.10.70.0 0.0.0.255
 remark Explicitly permit required services to the server VLAN
 permit udp 10.10.20.0 0.0.0.255 host 10.10.80.10 eq 53
 permit udp 10.10.25.0 0.0.0.255 host 10.10.80.10 eq 53
 permit tcp 10.10.20.0 0.0.0.255 host 10.10.80.10 eq 80
 permit tcp 10.10.25.0 0.0.0.255 host 10.10.80.10 eq 80
 permit icmp 10.10.20.0 0.0.0.255 host 10.10.80.10
 permit icmp 10.10.25.0 0.0.0.255 host 10.10.80.10
 remark Permit everything else, e.g. Internet access via NAT at R-EDGE
 permit ip any any
!

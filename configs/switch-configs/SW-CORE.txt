! =============================================================
! SW-CORE  -  Layer-3 Core/Distribution Switch
! Role: Inter-VLAN routing (SVIs), DHCP server for all department
!       VLANs, spanning-tree root for every VLAN
! Platform: Cisco Catalyst 3560-24PS (multilayer switch)
!
! NOTE ON SCOPE (verified directly in Packet Tracer 9.0): the
! Student/Wireless restriction ACL was originally designed to be
! applied here, on the Vlan20/Vlan25 SVIs ("ip access-group
! STUDENT-RESTRICT in"). Tested directly on this device: the
! command is accepted with no error, but never actually takes
! effect - confirmed three ways (show ip interface <vlan>, show
! running-config, and a live ping test that still got through).
! This is a genuine simulated-3560 limitation in this Packet
! Tracer build, not a configuration mistake. The ACL enforcement
! was moved to port ACLs on the access switches instead (verified
! working there - see SW-STUDENT.txt) and the STUDENT-RESTRICT
! ACL definition below is kept for reference/documentation only;
! it is not applied to any interface on this device. Full
! verification notes: docs/security.md section 6.
! =============================================================

hostname SW-CORE
!
ip routing
!
no ip domain-lookup
ip domain-name college.local
enable secret Cl@ssC0llege!Core
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY - College Core Switch (SW-CORE)
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Core
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Core
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
snmp-server community COLLEGE-RO RO
!

! VLAN Database

spanning-tree mode pvst
spanning-tree vlan 10,20,25,30,40,50,60,70,80,99 priority 4096

! Trunk uplinks to access switches

 switchport mode trunk
 switchport nonegotiate
 no shutdown
!
interface FastEthernet0/1
 description TRUNK-TO-SW-ACCOUNTS

!
interface FastEthernet0/2

interface FastEthernet0/4

 switchport trunk encapsulation dot1q
 switchport trunk native vlan 99

 switchport mode trunk

!
! Unused ports hardened
! NOTE: 3560-24PS has 24x FastEthernet (Fa0/1-24) + 2x GigabitEthernet (Gi0/1-2) only.
! Fa0/1-0/4 are used above for access-switch trunks; Fa0/5-24 are unused.
interface range FastEthernet0/5-24
 switchport mode access
 switchport access vlan 999

! -------------------------------------------------------------
! Reference copy only - NOT applied on this device (see NOTE at
! top of file). The working copy of this ACL, actually applied
! to interfaces, lives on SW-STUDENT.txt.
! -------------------------------------------------------------
ip access-list extended STUDENT-RESTRICT
 remark Deny Students/Wireless -> sensitive Accounts & Management subnets
 deny   ip 10.10.20.0 0.0.0.255 10.10.60.0 0.0.0.255
 deny   ip 10.10.20.0 0.0.0.255 10.10.70.0 0.0.0.255
 deny   ip 10.10.25.0 0.0.0.255 10.10.60.0 0.0.0.255
 deny   ip 10.10.25.0 0.0.0.255 10.10.70.0 0.0.0.255
 remark Explicitly permit required services to the server VLAN
 permit udp 10.10.20.0 0.0.0.255 host 10.10.80.10 eq 53
 permit udp 10.10.25.0 0.0.0.255 host 10.10.80.10 eq 53
 permit tcp 10.10.20.0 0.0.0.255 host 10.10.80.10 eq 80
 permit tcp 10.10.25.0 0.0.0.255 host 10.10.80.10 eq 80
 permit icmp 10.10.20.0 0.0.0.255 host 10.10.80.10
 permit icmp 10.10.25.0 0.0.0.255 host 10.10.80.10
 remark Permit everything else, e.g. Internet access via NAT at R-EDGE
 permit ip any any
!

! =============================================================
! R-EDGE  -  Campus Border Router
! Role: NAT/PAT, IOS Firewall (CBAC), Site-to-Site IPsec VPN peer,
!       default routing between campus, remote office, and Internet
! Platform: Cisco 2911 (or equivalent ISR in Packet Tracer)
! =============================================================

hostname R-EDGE
!
no ip domain-lookup
ip domain-name college.local
!
enable secret Cl@ssC0llege!Edge
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY
 This system is the property of the College IT Department.
 Unauthorized access, use, or modification is prohibited and
 may be subject to disciplinary action and/or legal prosecution.
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Edge
!
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Edge
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
snmp-server community COLLEGE-RO RO
!
! -------------------------------------------------------------
! Interfaces
! -------------------------------------------------------------
interface GigabitEthernet0/0
 description LINK-TO-SW-CORE (campus LAN transit)
 ip address 10.10.1.2 255.255.255.252
 ip nat inside
 ip inspect FW-INSPECT in
 no shutdown
!
interface GigabitEthernet0/1
 description WAN-LINK-TO-ISP-RTR (Internet edge)
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 ip access-group WAN-IN in
 ip inspect FW-INSPECT out
 crypto map CMAP-VPN
 no shutdown
!
! -------------------------------------------------------------
! Static Routing
! -------------------------------------------------------------
ip route 10.10.0.0 255.255.0.0 10.10.1.1
ip route 10.20.10.0 255.255.255.0 198.51.100.1
ip route 0.0.0.0 0.0.0.0 198.51.100.1
!
! -------------------------------------------------------------
! NAT / PAT (overload) - campus -> Internet only
! Remote-office (VPN) traffic is explicitly exempted from NAT
! -------------------------------------------------------------
ip access-list extended NAT-EXEMPT-AND-TRANSLATE
 deny   ip 10.10.0.0 0.0.255.255 10.20.10.0 0.0.0.255
 permit ip 10.10.0.0 0.0.255.255 any
!
ip nat inside source list NAT-EXEMPT-AND-TRANSLATE interface GigabitEthernet0/1 overload
!
! -------------------------------------------------------------
! IOS Firewall - CBAC (stateful inspection concept)
! -------------------------------------------------------------
ip inspect name FW-INSPECT tcp
ip inspect name FW-INSPECT udp
ip inspect name FW-INSPECT icmp
!
ip access-list extended WAN-IN
 remark Allow established VPN negotiation from remote site / ISP
 permit udp host 198.51.100.1 any eq isakmp
 permit esp host 198.51.100.1 any
 remark Allow ICMP replies in for troubleshooting (ping out works)
 permit icmp any any echo-reply
 permit icmp any any time-exceeded
 permit icmp any any port-unreachable
 remark Deny all other unsolicited inbound traffic from the Internet
 deny ip any any log
!
! -------------------------------------------------------------
! Site-to-Site IPsec VPN to R-REMOTE (Remote Admin Office)
! -------------------------------------------------------------
crypto isakmp policy 10
 encryption aes 256
 hash sha256
 authentication pre-share
 group 14
 lifetime 3600
!
crypto isakmp key C0llege-VPN-PSK-2026 address 198.51.100.6
!
crypto ipsec transform-set TS-COLLEGE esp-aes 256 esp-sha256-hmac
 mode tunnel
!
access-list 150 permit ip 10.10.0.0 0.0.255.255 10.20.10.0 0.0.0.255
!
crypto map CMAP-VPN 10 ipsec-isakmp
 set peer 198.51.100.6
 set transform-set TS-COLLEGE
 match address 150
!
end

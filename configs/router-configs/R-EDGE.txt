! =============================================================
! R-EDGE  -  Campus Border Router
! Role: NAT/PAT, perimeter ACL firewall, default routing between
!       campus, remote office, and Internet
! Platform: Cisco 2911 (or equivalent ISR in Packet Tracer)
!
! NOTE ON SCOPE (verified directly in Packet Tracer 9.0):
! This design originally also specified an IOS CBAC firewall
! ("ip inspect ...") and a site-to-site IPsec VPN to R-REMOTE
! ("crypto isakmp / crypto ipsec / crypto map"). Both command sets
! were tested directly on this router in Packet Tracer 9.0 and
! returned "% Invalid input detected" / "% Unrecognized command" -
! confirmed NOT supported by this platform's simulated IOS, even
! after enabling "license boot module c2900 technology-package
! securityk9" and reloading. Per project policy, unsupported
! functionality is not faked here. See docs/security.md and
! docs/vpn.md for the full verification notes and the real-world
! concept explanation. Perimeter filtering is still provided by
! the WAN-IN extended ACL below (stateless, but genuinely applied
! and verified working).
! =============================================================

hostname R-EDGE
!
no ip domain-lookup
ip domain-name college.local
!
enable secret Cl@ssC0llege!Edge
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY
 This system is the property of the College IT Department.
 Unauthorized access, use, or modification is prohibited and
 may be subject to disciplinary action and/or legal prosecution.
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Edge
!
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Edge
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
snmp-server community COLLEGE-RO RO
!
! -------------------------------------------------------------
! Interfaces
! -------------------------------------------------------------
interface GigabitEthernet0/0
 description LINK-TO-SW-CORE (campus LAN transit)
 ip address 10.10.1.2 255.255.255.252
 ip nat inside
 no shutdown
!
interface GigabitEthernet0/1
 description WAN-LINK-TO-ISP-RTR (Internet edge)
 ip address 198.51.100.2 255.255.255.252
 ip nat outside
 ip access-group WAN-IN in
 no shutdown
!
! -------------------------------------------------------------
! Static Routing
! -------------------------------------------------------------
ip route 10.10.0.0 255.255.0.0 10.10.1.1
ip route 10.20.10.0 255.255.255.0 198.51.100.1
ip route 0.0.0.0 0.0.0.0 198.51.100.1
!
! -------------------------------------------------------------
! NAT / PAT (overload) - campus -> Internet only.
! Campus-to-remote-office traffic is exempted from NAT so it
! keeps its real source address across the WAN (plain routed
! inter-site traffic - see NOTE above regarding VPN encryption).
! -------------------------------------------------------------
ip access-list extended NAT-EXEMPT-AND-TRANSLATE
 deny   ip 10.10.0.0 0.0.255.255 10.20.10.0 0.0.0.255
 permit ip 10.10.0.0 0.0.255.255 any
!
ip nat inside source list NAT-EXEMPT-AND-TRANSLATE interface GigabitEthernet0/1 overload
!
! -------------------------------------------------------------
! Perimeter ACL (stateless packet filter) - WAN-facing firewall
! -------------------------------------------------------------
ip access-list extended WAN-IN
 remark Allow ICMP replies in for troubleshooting (ping out works)
 permit icmp any any echo-reply
 permit icmp any any time-exceeded
 permit icmp any any port-unreachable
 remark Deny all other unsolicited inbound traffic from the Internet
 deny ip any any log
!
end

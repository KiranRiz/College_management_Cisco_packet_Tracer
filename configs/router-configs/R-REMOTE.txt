! =============================================================
! R-REMOTE  -  Remote Admin Office Branch Router
! Role: Default gateway for the remote site, routed back to the
!       main campus over the WAN (see NOTE below on VPN scope)
! Platform: Cisco 2901 (or equivalent ISR in Packet Tracer)
!
! NOTE ON SCOPE: a site-to-site IPsec VPN to R-EDGE was designed
! and tested here but Packet Tracer 9.0's simulated IOS rejects
! all "crypto isakmp / crypto ipsec / crypto map" commands as
! unrecognized on this platform (verified directly, including
! after enabling the securityk9 license and reloading). Per
! project policy, this is not faked - the remote site is reached
! by plain static routing instead, and VPN is documented as a
! concept in docs/vpn.md. See docs/security.md for the full
! verification notes.
! =============================================================

hostname R-REMOTE
!
no ip domain-lookup
ip domain-name college.local
!
enable secret Cl@ssC0llege!Remote
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY - College Remote Admin Office Router
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Remote
!
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Remote
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
!
! -------------------------------------------------------------
! Interfaces
! -------------------------------------------------------------
interface GigabitEthernet0/0
 description WAN-LINK-TO-ISP-RTR
 ip address 198.51.100.6 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/1
 description LINK-TO-SW-REMOTE (remote office LAN)
 ip address 10.20.10.1 255.255.255.0
 no shutdown
!
! -------------------------------------------------------------
! Routing - single default route (branch office)
! -------------------------------------------------------------
ip route 0.0.0.0 0.0.0.0 198.51.100.5
!
! -------------------------------------------------------------
! DHCP for the remote office LAN
! -------------------------------------------------------------
ip dhcp excluded-address 10.20.10.1 10.20.10.9
ip dhcp pool REMOTE-OFFICE-POOL
 network 10.20.10.0 255.255.255.0
 default-router 10.20.10.1
 dns-server 10.10.80.10
 domain-name college.local
!
end

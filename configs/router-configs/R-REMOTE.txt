! =============================================================
! R-REMOTE  -  Remote Admin Office Branch Router
! Role: Default gateway for the remote site, IPsec VPN peer back
!       to the main campus (R-EDGE)
! Platform: Cisco 2901 (or equivalent ISR in Packet Tracer)
! =============================================================

hostname R-REMOTE
!
no ip domain-lookup
ip domain-name college.local
!
enable secret Cl@ssC0llege!Remote
service password-encryption
!
banner motd #
***************************************************************
 AUTHORIZED ACCESS ONLY - College Remote Admin Office Router
***************************************************************
#
!
username netadmin privilege 15 secret N3tAdm1n@Remote
!
crypto key generate rsa modulus 2048
ip ssh version 2
!
line con 0
 password C0ns0le@Remote
 login
line vty 0 4
 login local
 transport input ssh
!
logging buffered 16384 informational
!
! -------------------------------------------------------------
! Interfaces
! -------------------------------------------------------------
interface GigabitEthernet0/0
 description WAN-LINK-TO-ISP-RTR
 ip address 198.51.100.6 255.255.255.252
 crypto map CMAP-VPN
 no shutdown
!
interface GigabitEthernet0/1
 description LINK-TO-SW-REMOTE (remote office LAN)
 ip address 10.20.10.1 255.255.255.0
 no shutdown
!
! -------------------------------------------------------------
! Routing - single default route (branch office)
! -------------------------------------------------------------
ip route 0.0.0.0 0.0.0.0 198.51.100.5
!
! -------------------------------------------------------------
! DHCP for the remote office LAN
! -------------------------------------------------------------
ip dhcp excluded-address 10.20.10.1 10.20.10.9
ip dhcp pool REMOTE-OFFICE-POOL
 network 10.20.10.0 255.255.255.0
 default-router 10.20.10.1
 dns-server 10.10.80.10
 domain-name college.local
!
! -------------------------------------------------------------
! Site-to-Site IPsec VPN back to R-EDGE (main campus)
! -------------------------------------------------------------
crypto isakmp policy 10
 encryption aes 256
 hash sha256
 authentication pre-share
 group 14
 lifetime 3600
!
crypto isakmp key C0llege-VPN-PSK-2026 address 198.51.100.2
!
crypto ipsec transform-set TS-COLLEGE esp-aes 256 esp-sha256-hmac
 mode tunnel
!
access-list 150 permit ip 10.20.10.0 0.0.0.255 10.10.0.0 0.0.255.255
!
crypto map CMAP-VPN 10 ipsec-isakmp
 set peer 198.51.100.2
 set transform-set TS-COLLEGE
 match address 150
!
end
